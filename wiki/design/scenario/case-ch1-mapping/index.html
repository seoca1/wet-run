    // story.html의 chapter-flow-viewer가 동적으로 로드
fetch('../data/story/arcs/chapter_flow.json')
  .then(r => r.json())
  .then(data => {
    // 케이/실/카스 캐릭터별 챕터 흐름 렌더링
    // Phase, Combat, Gain, Cutscene 정보 포함
  });

    연동 파일: - prototype/src/wet_run/engine/chapter_cutscene.py — PhaseData에 combat, gain, loss 필드 추가 - prototype/data/story/arcs/case_arc.json — phases에 story-specific 메타데이터 추가 - dashboard/story.html — 챕터 흐름 뷰어 섹션 추가 - .github/workflows/pages.yml — dashboard/data/ 디렉토리 복사 추가
